Hooks: a script that says no



  
# .git/hooks/pre-commit
#!/usr/bin/env bash
./test.sh >/dev/null      || exit 1
git diff --cached |
  grep -q '^+.*TODO'        && exit 1
exit 0

  

    
exit 0the commit happens

    
exit 1it does not
